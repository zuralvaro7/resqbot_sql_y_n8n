CREATE TABLE paciente (
    numpaci INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellid VARCHAR(150) NOT NULL,
    telefono VARCHAR(20),
    email VARCHAR(150),
    direccion VARCHAR(255),
    persconta VARCHAR(150),
    telconta VARCHAR(20),
    emailconta VARCHAR(150),
    dirconta VARCHAR(255)
);

CREATE TABLE conversacion (
    numchat INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    numpaci INT NOT NULL,
    msgent TEXT NOT NULL,
    tsent TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    tssal TIMESTAMP WITH TIME ZONE,
    alerta BOOLEAN DEFAULT FALSE,
    priorid INT,
    imagen BYTEA,
    audio BYTEA,
    respuesta TEXT,

    CONSTRAINT fk_conv_pac
        FOREIGN KEY (numpaci)
        REFERENCES paciente(numpaci)
        ON DELETE CASCADE
);

CREATE TABLE alerta (
    idalert INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    numpaci INT NOT NULL,
    numchat INT NOT NULL,
    msgent TEXT NOT NULL,
    motivo VARCHAR(255) NOT NULL,

    CONSTRAINT fk_alert_pac
        FOREIGN KEY (numpaci)
        REFERENCES paciente(numpaci)
        ON DELETE CASCADE,

    CONSTRAINT fk_alert_chat
        FOREIGN KEY (numchat)
        REFERENCES conversacion(numchat)
        ON DELETE CASCADE
);

CREATE TABLE usuario (
    iduser INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    numpaci INT,
    usuario VARCHAR(100) NOT NULL UNIQUE,
    passhash VARCHAR(255) NOT NULL,
    rol VARCHAR(50) NOT NULL,

    CONSTRAINT fk_user_pac
        FOREIGN KEY (numpaci)
        REFERENCES paciente(numpaci)
        ON DELETE CASCADE
);

CREATE INDEX idx_conv_numpaci ON conversacion(numpaci);
CREATE INDEX idx_alert_numpaci ON alerta(numpaci);
CREATE INDEX idx_alert_chat ON alerta(numchat);
CREATE INDEX idx_user_numpaci ON usuario(numpaci);
